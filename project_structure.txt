=== ./agentic-domain/src/main/java/com/im/agentic/domain/transaction/entity/Transaction.java ===
package com.im.agentic.domain.transaction.entity;

import com.im.agentic.domain.common.BaseTimeEntity;
import com.im.agentic.domain.goal.entity.Goal;
import com.im.agentic.domain.mission.entity.Mission;
import jakarta.persistence.*;
import lombok.*;

@Entity
@Table(name = "transactions")
@Getter
@Builder
@NoArgsConstructor(access = AccessLevel.PROTECTED)
@AllArgsConstructor
public class Transaction extends BaseTimeEntity {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "goal_id", nullable = false)
    private Goal goal;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "mission_id")
    private Mission mission;

    @Column(name = "base_saving_amount", nullable = false)
    private Long baseSavingAmount;

    @Column(name = "bonus_saving_amount", nullable = false)
    private Long bonusSavingAmount;

    @Column(name = "total_amount", nullable = false)
    private Long totalAmount;

    @Enumerated(EnumType.STRING)
    @Column(name = "status", nullable = false)
    private TransactionStatus status;
}

package com.im.agentic.domain.mydata.entity;

import com.im.agentic.domain.common.BaseTimeEntity;
import com.im.agentic.domain.user.entity.User;
import jakarta.persistence.*;
import lombok.*;
import org.hibernate.annotations.JdbcTypeCode;
import org.hibernate.type.SqlTypes;

import java.time.LocalDate;
import java.time.LocalDateTime;

@Entity
@Table(name = "my_data_caches")
@Getter
@Builder
@NoArgsConstructor(access = AccessLevel.PROTECTED)
@AllArgsConstructor
public class MyDataCache extends BaseTimeEntity {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @OneToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "user_id", nullable = false, unique = true)

    @Builder.Default
    @Column(name = "current_amount", nullable = false)

    @Column(name = "withdrawal_condition", columnDefinition = "jsonb")
    private String withdrawalCondition;

    @Column(name = "lock_id")
    private String lockId;

    public void updateProgress(Long amount) {
        this.currentAmount += amount;
        this.achievementRate = (double) this.currentAmount / this.targetAmount * 100;
        if (this.achievementRate >= 100.0) {
            this.status = GoalStatus.COMPLETED;
        }
    }

    public void updateGoalDetails(String itemName, Long targetAmount, String itemImageUrl) {
        this.itemName = itemName;
        this.targetAmount = targetAmount;
        this.itemImageUrl = itemImageUrl;
    }

        this.status = GoalStatus.CANCELLED;
    }

        this.status = GoalStatus.COMPLETED;
    }

        this.withdrawalCondition = withdrawalCondition;
    }

    }

async def ollama_embed(text: str) -> list[float]:
    try:
        response = await http_client.post(
            f"{OLLAMA_URL}/api/embeddings",
            json={"model": "nomic-embed-text", "prompt": text}
        )
        return response.json().get("embedding", [])
    except Exception as e:
        logger.error(f"임베딩 생성 실패: {e}")
        return []

class SpendingRecord(BaseModel):
    category: str
    amount: int
    merchantName: str
    date: str

class SpendingAnalysisRequest(BaseModel):
    userId: int
    spendings: list[SpendingRecord]

class SpendingAnalysisResponse(BaseModel):
    categoryExpenses: dict[str, int]
    totalExpenses: int
    dailyAvailableAmount: int
    topCategory: str

@app.post("/agent/analyze-spending", response_model=SpendingAnalysisResponse)
async def analyze_spending(request: SpendingAnalysisRequest):
    logger.info(f"[SpendingAnalysisAgent] userId={request.userId}, records={len(request.spendings)}")

    category_expenses = {}
    for record in request.spendings:
        category_expenses[record.category] = \
            category_expenses.get(record.category, 0) + record.amount

    total = sum(category_expenses.values())
    income = max(3000000, total + 600000)
    daily = (income - total) // 30
    top = max(category_expenses, key=category_expenses.get) if category_expenses else "식비"

    # 1. 20개 레코드 임베딩 병렬 생성 (HTTP 커넥션 재사용)
    async def get_embedding(rec):
        content = f"{rec.category} {rec.merchantName} {rec.amount}원"
        emb = await ollama_embed(content)
        return (rec, emb)

    results = await asyncio.gather(*(get_embedding(r) for r in request.spendings[:20]))

    # 2. 단일 커넥션으로 벌크 인서트 처리
    try:
        conn = get_db_conn()
        cur = conn.cursor()
        for record, embedding in results:
            if not embedding:
                continue
            content = f"{record.category} {record.merchantName} {record.amount}원"
            cur.execute("""
                INSERT INTO spending_embeddings (user_id, category, merchant_name, amount, embedding, content, created_at, updated_at)
                VALUES (%s, %s, %s, %s, %s::vector, %s, NOW(), NOW())
            """, (request.userId, record.category, record.merchantName, record.amount, str(embedding), content))
        conn.commit()
        conn.close()
        logger.info(f"[SpendingAnalysisAgent] 임베딩 저장 완료: {len(results)}건")
    except Exception as e:
        logger.error(f"[SpendingAnalysisAgent] 벌크 임베딩 저장 실패: {e}")

    logger.info(f"[SpendingAnalysisAgent] 완료: {list(category_expenses.keys())}")
    return SpendingAnalysisResponse(
        categoryExpenses=category_expenses,
        totalExpenses=total,
        dailyAvailableAmount=daily,
        topCategory=top
    )

async def ollama_generate(prompt: str) -> str:
    async with httpx.AsyncClient(timeout=180.0) as client:
        response = await client.post(
            f"{OLLAMA_URL}/api/generate",
            json={"model": OLLAMA_MODEL, "prompt": prompt, "stream": False, "format": "json"}
        )
        return response.json().get("response", "")

async def ollama_embed(text: str) -> list[float]:
    async with httpx.AsyncClient(timeout=30.0) as client:
        response = await client.post(
            f"{OLLAMA_URL}/api/embeddings",
            json={"model": "nomic-embed-text", "prompt": text}
        )
        return response.json().get("embedding", [])

def search_similar_spending(user_id: int, query_embedding: list, top_k: int = 3) -> list:
    try:
        conn = get_db_conn()
        cur = conn.cursor()
        cur.execute("""
            SELECT category, merchant_name, amount
            FROM spending_embeddings
            WHERE user_id = %s
            ORDER BY embedding <=> %s::vector
            LIMIT %s
        """, (user_id, str(query_embedding), top_k))
        rows = cur.fetchall()
        conn.close()
        return [{"category": r[0], "merchantName": r[1], "amount": r[2]} for r in rows]
    except Exception as e:
        logger.error(f"pgvector 검색 실패: {e}")
        return []

def clean_json(response: str) -> str:
    response = response.strip()
    if response.startswith("```json"):
        response = response[7:]
    elif response.startswith("```"):
        response = response[3:]
    if response.endswith("```"):
        response = response[:-3]
    return response.strip()

class GoalParsingRequest(BaseModel):
    userId: int
    rawInput: str

class GoalParsingResponse(BaseModel):
    itemName: str
    price: int
    category: str

@app.post("/agent/parse-goal", response_model=GoalParsingResponse)
async def parse_goal(request: GoalParsingRequest):
    logger.info(f"[GoalParsingAgent] userId={request.userId}, input={request.rawInput}")

    query_embedding = await ollama_embed(request.rawInput)
    similar = search_similar_spending(request.userId, query_embedding, top_k=3)
    context = "\n".join([f"- {s['category']} {s['merchantName']}: {s['amount']}원" for s in similar]) or "없음"

    prompt = f"""당신은 금융 목표 파싱 AI입니다.
사용자의 입력에서 물품명과 가격을 추출해 반드시 아래 JSON 형식으로만 응답하세요.
{{"itemName": "물품명", "price": 숫자, "category": "카테고리"}}
가격을 모르면 price를 0으로 설정하세요.

[참고: 유사 소비 내역]
{context}

[입력]
{request.rawInput}"""

    logger.info(f"[GoalParsingAgent] Prompt length: {len(prompt)} chars")
    raw = await ollama_generate(prompt)
    logger.info(f"[GoalParsingAgent] 응답: {raw}")

    try:
        data = json.loads(clean_json(raw))
        return GoalParsingResponse(
            itemName=data.get("itemName", request.rawInput),
            price=int(data.get("price", 0)),
            category=data.get("category", "기타")
        )
    except Exception as e:
        logger.error(f"[GoalParsingAgent] 파싱 실패: {e}")
        return GoalParsingResponse(itemName=request.rawInput, price=0, category="기타")

from fastapi import FastAPI
from pydantic import BaseModel
import httpx
import json

from pgvector.psycopg2 import register_vector
import os
import logging

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

app = FastAPI(title="iM AgentiX — MissionGenerationAgent")

OLLAMA_URL = os.getenv("OLLAMA_BASE_URL", "http://host.docker.internal:11434")
OLLAMA_MODEL = os.getenv("OLLAMA_MODEL", "qwen2.5:1.5b")
DB_HOST = os.getenv("DB_HOST", "agentic-postgres")
DB_PORT = os.getenv("DB_PORT", "5432")
DB_NAME = os.getenv("DB_NAME", "agentic")
DB_USER = os.getenv("DB_USER", "agentic_user")
DB_PASS = os.getenv("DB_PASS", "agentic_password")

def get_db_conn():
    conn = psycopg2.connect(
        host=DB_HOST, port=DB_PORT,
        dbname=DB_NAME, user=DB_USER, password=DB_PASS
    )
    register_vector(conn)
    return conn

async def ollama_generate(prompt: str) -> str:
    async with httpx.AsyncClient(timeout=180.0) as client:
        response = await client.post(
            f"{OLLAMA_URL}/api/generate",
            json={"model": OLLAMA_MODEL, "prompt": prompt, "stream": False, "format": "json"}
        )
        return response.json().get("response", "")

async def ollama_embed(text: str) -> list[float]:
    async with httpx.AsyncClient(timeout=30.0) as client:
        response = await client.post(
            f"{OLLAMA_URL}/api/embeddings",
            json={"model": "nomic-embed-text", "prompt": text}
        )
        return response.json().get("embedding", [])

def search_similar_spending(user_id: int, query_embedding: list, top_k: int = 3) -> list:
    try:
        conn = get_db_conn()
        cur = conn.cursor()
        cur.execute("""
            SELECT category, merchant_name, amount
            FROM spending_embeddings
            WHERE user_id = %s
            ORDER BY embedding <=> %s::vector
            LIMIT %s
        """, (user_id, str(query_embedding), top_k))
        rows = cur.fetchall()
        conn.close()
        return [{"category": r[0], "merchantName": r[1], "amount": r[2]} for r in rows]
    except Exception as e:
        logger.error(f"소비 검색 실패: {e}")
        return []

def search_similar_missions(query_embedding: list, top_k: int = 3) -> list:
    try:
        conn = get_db_conn()
        cur = conn.cursor()
        cur.execute("""
            SELECT category, mission_type, limit_amount, success_rate
            FROM mission_cases
            ORDER BY embedding <=> %s::vector
            LIMIT %s
        """, (str(query_embedding), top_k))
        rows = cur.fetchall()
        conn.close()
        return [{"category": r[0], "missionType": r[1],
                 "limitAmount": r[2], "successRate": r[3]} for r in rows]
    except Exception as e:
        logger.error(f"미션 검색 실패: {e}")
        return []

def save_mission_case(category: str, mission_type: str,
                       limit_amount: int, success_rate: float, embedding: list):
    try:
        conn = get_db_conn()
        cur = conn.cursor()
        content = f"{category} {mission_type} 미션 한도 {limit_amount}원 성공률 {success_rate*100:.0f}%"
        cur.execute("""
            INSERT INTO mission_cases (category, mission_type, limit_amount, success_rate, embedding, content, created_at, updated_at)
            VALUES (%s, %s, %s, %s, %s::vector, %s, NOW(), NOW())
        """, (category, mission_type, limit_amount, success_rate, str(embedding), content))
        conn.commit()
        conn.close()
    except Exception as e:
        logger.error(f"미션 사례 저장 실패: {e}")

def clean_json(response: str) -> str:
    response = response.strip()
    if response.startswith("```json"):
        response = response[7:]
    elif response.startswith("```"):
        response = response[3:]
    if response.endswith("```"):
        response = response[:-3]
    return response.strip()

class MissionGenerationRequest(BaseModel):
    userId: int
    categoryExpenses: dict[str, int]
    dailyAvailableAmount: int
    goalItemName: str

class MissionGenerationResponse(BaseModel):
    restraintCategory: str
    restraintLimitAmount: int
    restraintContributionRate: float
    savingTransferAmount: int
    savingContributionRate: float

@app.post("/agent/generate-mission", response_model=MissionGenerationResponse)
async def generate_mission(request: MissionGenerationRequest):
    logger.info(f"[MissionGenerationAgent] userId={request.userId}, goal={request.goalItemName}")

    query = f"소비: {request.categoryExpenses}, 가용: {request.dailyAvailableAmount}원, 목표: {request.goalItemName}"
    embedding = await ollama_embed(query)

    similar_spending = search_similar_spending(request.userId, embedding, top_k=3)
    similar_missions = search_similar_missions(embedding, top_k=3)

    spending_ctx = "\n".join([
        f"- {s['category']} {s['merchantName']}: {s['amount']}원"
        for s in similar_spending
    ]) or "없음"

    mission_ctx = "\n".join([
        f"- [{m['missionType']}] {m['category']}, 성공률 {m['successRate']*100:.0f}%"
        for m in similar_missions
    ]) or "없음"

    prompt = f"""당신은 금융 미션 생성 AI입니다.
반드시 아래 JSON 형식으로만 응답하세요. 다른 텍스트 없이 JSON만 출력하세요.
{{
  "restraint": {{"categoryName": "카테고리명", "limitAmount": 숫자, "contributionRate": 소수}},
  "saving": {{"transferAmount": 숫자, "contributionRate": 소수}}
}}

[유사 소비 내역]
{spending_ctx}

[성공 미션 사례]
{mission_ctx}

소비 패턴: {request.categoryExpenses}
가용 금액: {request.dailyAvailableAmount}원
목표: {request.goalItemName}"""

    logger.info(f"[MissionGenerationAgent] Prompt length: {len(prompt)} chars")
    raw = await ollama_generate(prompt)
    logger.info(f"[MissionGenerationAgent] RAG 적용 완료, 소비 {len(similar_spending)}건, 미션 {len(similar_missions)}건")

    try:
        data = json.loads(clean_json(raw))
        r = data["restraint"]
        s = data["saving"]
        result = MissionGenerationResponse(
            restraintCategory=r["categoryName"],
            restraintLimitAmount=int(r["limitAmount"]),
            restraintContributionRate=float(r["contributionRate"]),
            savingTransferAmount=int(s["transferAmount"]),
            savingContributionRate=float(s["contributionRate"])
        )
        # 성공 미션 사례 저장
        emb = await ollama_embed(f"{r['categoryName']} RESTRAINT 미션")
        if emb:
            save_mission_case(r["categoryName"], "RESTRAINT", int(r["limitAmount"]), 0.7, emb)
        return result
    except Exception as e:
        logger.error(f"[MissionGenerationAgent] 파싱 실패: {e}")
        return MissionGenerationResponse(
            restraintCategory="식비", restraintLimitAmount=10000,
            restraintContributionRate=1.0, savingTransferAmount=5000,
            savingContributionRate=0.5
        )

def health():
    return {"status": "ok", "agent": "MissionGenerationAgent", "port": 8003}

class ExecutionRequest(BaseModel):
    userId: int
    goalId: int
    amount: int
    missionType: str
    categoryName: str | None = None
    limitAmount: int | None = None
    actualSpending: int | None = None

class ExecutionResponse(BaseModel):
    success: bool
    shouldTransfer: bool
    savedAmount: int
    message: str

@app.post("/agent/execute", response_model=ExecutionResponse)
async def execute_mission(request: ExecutionRequest):
    logger.info(f"[ExecutionAgent] userId={request.userId}, type={request.missionType}")

    if request.missionType == "RESTRAINT":
        actual = request.actualSpending or 0
        limit = request.limitAmount or 0
        if actual <= limit:
            saved = limit - actual
            logger.info(f"[ExecutionAgent] RESTRAINT 성공: saved={saved}원")
            return ExecutionResponse(
                success=True, shouldTransfer=True, savedAmount=saved,
                message=f"{request.categoryName} 절제 성공! {saved}원 절약"
            )
        else:
            logger.info(f"[ExecutionAgent] RESTRAINT 실패: actual={actual}, limit={limit}")
            return ExecutionResponse(
                success=False, shouldTransfer=False, savedAmount=0,
                message=f"{request.categoryName} 한도 초과. 실제: {actual}원, 한도: {limit}원"
            )
    else:  # SAVING
        logger.info(f"[ExecutionAgent] SAVING 성공: amount={request.amount}원")
        return ExecutionResponse(
            success=True, shouldTransfer=True, savedAmount=request.amount,
            message=f"저축 미션 성공! {request.amount}원 이체"
        )

    global _mlx_model, _mlx_tokenizer

        elif os.path.exists(ADAPTER_PATH):
            _mlx_model, _mlx_tokenizer = load("mlx-community/Qwen2.5-1.5B-Instruct-4bit", adapter_path=ADAPTER_PATH)
        else:
            _mlx_model, _mlx_tokenizer = load("mlx-community/Qwen2.5-1.5B-Instruct-4bit")
    return _mlx_model, _mlx_tokenizer

OLLAMA_AVAILABLE = None

def check_ollama_available():
    global OLLAMA_AVAILABLE
    if OLLAMA_AVAILABLE is not None:
        return OLLAMA_AVAILABLE
    try:
        with httpx.Client(timeout=1.0) as client:
            response = client.get(f"{OLLAMA_URL}/api/tags")
            if response.status_code == 200:
                OLLAMA_AVAILABLE = True
                print("[Ollama] 서비스 감지 성공. Ollama를 우선 사용합니다.")
                return True
    except Exception:
        pass
    OLLAMA_AVAILABLE = False
    print("[Ollama] 서비스 응답 없음. 로컬 MLX 모드로 직행합니다.")
    return False

def generate_ollama(prompt):

    except Exception as e:

    return None

def clean_json(text):
    text = text.strip()
    if text.startswith("```json"):
        text = text[7:]
    elif text.startswith("```"):
        text = text[3:]
    if text.endswith("```"):
        text = text[:-3]
    text = text.strip()
    match = re.search(r"(\{.*\})", text, re.DOTALL)
    if match:
        return match.group(1)
    return text

def parse_to_json(text):
    try:
        cleaned = clean_json(text)
        return json.loads(cleaned)
    except Exception:
        return None

def get_stats(lst):
    if not lst:
        return 0.0, 0.0, 0.0, 0.0
    n = len(lst)
    mean = sum(lst) / n
    var = sum((x - mean) ** 2 for x in lst) / n
    std = math.sqrt(var)
    return mean, min(lst), max(lst), std

def run_quality_test():
    if not os.path.exists(MISSION_DATASET_PATH):
        print(f"오류: 미션 데이터셋이 존재하지 않습니다: {MISSION_DATASET_PATH}")
        return

    print("미션 데이터셋 로드 중...")
    with open(MISSION_DATASET_PATH, "r", encoding="utf-8") as f:
        samples = [json.loads(line) for line in f if line.strip()]

    # 무작위 50개 샘플 선택
    random.seed(24)
    test_samples = random.sample(samples, min(50, len(samples)))
    print(f"품질 테스트를 위해 {len(test_samples)}개의 샘플을 선정했습니다.")

    restraint_limits = []
    restraint_rates = []
    saving_amounts = []
    saving_rates = []
    parsing_failures = 0

    for idx, sample in enumerate(test_samples):
        prompt = sample["prompt"]
        raw_res = query_tuned_model(prompt)
        res_json = parse_to_json(raw_res)

        if res_json:
            try:
                # 절제 한도
                limit = res_json.get("restraint", {}).get("limitAmount", 0)
                rate_r = res_json.get("restraint", {}).get("contributionRate", 0.0)
                # 저축 이체
                transfer = res_json.get("saving", {}).get("transferAmount", 0)
                rate_s = res_json.get("saving", {}).get("contributionRate", 0.0)

                restraint_limits.append(float(limit))
                restraint_rates.append(float(rate_r))
                saving_amounts.append(float(transfer))
                saving_rates.append(float(rate_s))
            except Exception as e:
                print(f"[오류] 데이터 파싱 또는 타입 변환 실패: {e}")
                parsing_failures += 1
        else:
            parsing_failures += 1

        if (idx + 1) % 10 == 0:
            print(f"품질 테스트 진행률: {idx + 1}/{len(test_samples)}")

    # 통계 계산
    limit_mean, limit_min, limit_max, limit_std = get_stats(restraint_limits)
    rate_r_mean, rate_r_min, rate_r_max, rate_r_std = get_stats(restraint_rates)
    save_mean, save_min, save_max, save_std = get_stats(saving_amounts)
    rate_s_mean, rate_s_min, rate_s_max, rate_s_std = get_stats(saving_rates)

    successful_runs = len(test_samples) - parsing_failures

    report_content = f"""## 3단계: 미션 생성 품질 분포 테스트 (총 {len(test_samples)}회 테스트)

"""
    # 결과를 파일로 임시 저장

    pattern = {}
    for cat in random.sample(CATEGORIES, random.randint(3, 6)):
        if cat == "카페":
            pattern[cat] = random.randint(20000, 80000)
        elif cat == "식비":
            pattern[cat] = random.randint(80000, 300000)
        elif cat == "교통":
            pattern[cat] = random.randint(30000, 80000)
        elif cat == "배달음식":
            pattern[cat] = random.randint(50000, 200000)
        else:
            pattern[cat] = random.randint(10000, 100000)
    return pattern

def generate_mission_from_pattern(pattern, daily_available, goal_item):
    top_category = max(pattern, key=pattern.get)
    top_amount = pattern[top_category]
    limit = int(top_amount * random.uniform(0.5, 0.8) / 30)
    limit = max(3000, (limit // 1000) * 1000)

    saving = int(daily_available * random.uniform(0.2, 0.5))
    saving = max(1000, (saving // 1000) * 1000)

    return {
        "restraint": {
            "categoryName": top_category,
            "limitAmount": limit,
            "contributionRate": round(random.uniform(0.5, 2.0), 1)
        },
        "saving": {
            "transferAmount": saving,
            "contributionRate": round(random.uniform(0.3, 1.5), 1)
        }
    }

def generate_mission_dataset(n=300):
    data = []
    for _ in range(n):

        goal_item = random.choice(GOAL_ITEMS)

import json
import os
import random

SCRIPT_DIR = os.path.dirname(os.path.abspath(__file__))
DATA_DIR = os.path.join(SCRIPT_DIR, "data")

def load_jsonl(file_path):
    data = []
    with open(file_path, "r", encoding="utf-8") as f:
        for line in f:
            if line.strip():
                data.append(json.loads(line))
    return data

def save_jsonl(data, file_path):
    with open(file_path, "w", encoding="utf-8") as f:
        for item in data:
            f.write(json.dumps(item, ensure_ascii=False) + "\n")

def split_dataset():
    goal_parsing_path = os.path.join(DATA_DIR, "goal_parsing.jsonl")
    mission_gen_path = os.path.join(DATA_DIR, "mission_generation.jsonl")

    # 원본 파일이 없을 시 데이터셋 재생성 시도
    if not os.path.exists(goal_parsing_path) or not os.path.exists(mission_gen_path):
        print("원본 데이터셋 파일이 존재하지 않습니다. generate_dataset.py를 실행합니다...")
        import generate_dataset
        generate_dataset.generate_goal_parsing_dataset(500)
        generate_dataset.generate_mission_dataset(350)

    goal_data = load_jsonl(goal_parsing_path)
    mission_data = load_jsonl(mission_gen_path)

    # 데이터 순서 무작위화
    random.seed(42)
    random.shuffle(goal_data)
    random.shuffle(mission_data)

    # 파싱 500건: train 400 + valid 50 + golden 50
    goal_train = goal_data[0:400]
    goal_valid = goal_data[400:450]
    goal_golden = goal_data[450:500]

    # 미션 300(실제 350건 생성): train 250 + valid 50 + golden 50
    if len(mission_data) < 350:
        print(f"미션 데이터 개수({len(mission_data)})가 분할 요구량(350)보다 부족합니다. 재생성합니다...")
        import generate_dataset
        mission_data = generate_dataset.generate_mission_dataset(350)
        random.shuffle(mission_data)

    mission_train = mission_data[0:250]
    mission_valid = mission_data[250:300]
    mission_golden = mission_data[300:350]

    # 병합
    train_set = goal_train + mission_train
    valid_set = goal_valid + mission_valid
    golden_set = goal_golden + mission_golden

    # 무작위 셔플
    random.shuffle(train_set)
    random.shuffle(valid_set)
    random.shuffle(golden_set)

    # 저장
    save_jsonl(train_set, os.path.join(DATA_DIR, "train.jsonl"))
    save_jsonl(valid_set, os.path.join(DATA_DIR, "valid.jsonl"))
    save_jsonl(golden_set, os.path.join(DATA_DIR, "golden_eval.jsonl"))

import json

SCRIPT_DIR = os.path.dirname(os.path.abspath(__file__))
DATA_DIR = os.path.join(SCRIPT_DIR, "data")
GOLDEN_PATH = os.path.join(DATA_DIR, "golden_eval.jsonl")

OLLAMA_URL = "http://localhost:11434"
BASE_MODEL = "qwen2.5:1.5b"
TUNED_MODEL = "qwen2.5-tuned"
FUSED_MODEL_PATH = os.path.join(SCRIPT_DIR, "fused_model")
ADAPTER_PATH = os.path.join(os.path.dirname(SCRIPT_DIR), "adapters") # mlx_lm default save path

# MLX 로컬 모델 싱글톤 캐시 (폴백용)
_mlx_model = None
_mlx_tokenizer = None

def get_mlx_model():
    global _mlx_model, _mlx_tokenizer
    if _mlx_model is None:
        print("[Fallback] MLX 패키지를 이용해 로컬에서 직접 모델을 로드합니다...")
        from mlx_lm import load
        # Fused model이 존재하면 이를 우선 로드, 아니면 베이스 + 어댑터 로드
        if os.path.exists(FUSED_MODEL_PATH):
            print(f"[Fallback] Fused 모델 로드 중: {FUSED_MODEL_PATH}")
            _mlx_model, _mlx_tokenizer = load(FUSED_MODEL_PATH)
        elif os.path.exists(ADAPTER_PATH):
            print(f"[Fallback] 베이스 모델 + 어댑터 로드 중: {ADAPTER_PATH}")
            _mlx_model, _mlx_tokenizer = load("mlx-community/Qwen2.5-1.5B-Instruct-4bit", adapter_path=ADAPTER_PATH)
        else:
            print("[Fallback] 경고: 학습된 어댑터나 Fused 모델을 찾을 수 없습니다. 베이스 모델을 사용합니다.")
            _mlx_model, _mlx_tokenizer = load("mlx-community/Qwen2.5-1.5B-Instruct-4bit")
    return _mlx_model, _mlx_tokenizer

OLLAMA_AVAILABLE = None

def check_ollama_available():
    global OLLAMA_AVAILABLE
    if OLLAMA_AVAILABLE is not None:
        return OLLAMA_AVAILABLE
    try:
        with httpx.Client(timeout=1.0) as client:
            response = client.get(f"{OLLAMA_URL}/api/tags")
            if response.status_code == 200:
                OLLAMA_AVAILABLE = True
                print("[Ollama] 서비스 감지 성공. Ollama를 우선 사용합니다.")
                return True
    except Exception:
        pass
    OLLAMA_AVAILABLE = False
    print("[Ollama] 서비스 응답 없음. 로컬 MLX 모드로 직행합니다.")
    return False

def generate_ollama(model_name, prompt):
    if not check_ollama_available():
        return None
    try:
        with httpx.Client(timeout=10.0) as client:
            response = client.post(
                f"{OLLAMA_URL}/api/generate",
                json={"model": model_name, "prompt": prompt, "stream": False}
            )
            if response.status_code == 200:
                return response.json().get("response", "").strip()
    except Exception as e:
        print(f"[Ollama] {model_name} 호출 실패: {e}")
    return None

def generate_mlx(prompt):
    try:
        from mlx_lm import generate
        model, tokenizer = get_mlx_model()
        response = generate(model, tokenizer, prompt=prompt, max_tokens=256, verbose=False)
        return response.strip()
    except Exception as e:
        print(f"[MLX Fallback] 직접 추론 실패: {e}")
        return ""

def query_model(model_name, prompt):
    # 베이스 모델은 Ollama를 기본 호출하되, Ollama가 꺼져있으면 연산 없이 건너뜀 (크래시 방지)
    if model_name == BASE_MODEL:
        if not check_ollama_available():
            return ""
        res = generate_ollama(BASE_MODEL, prompt)
        return res if res else ""
    
    # 튜닝된 모델: Ollama 사용 가능하면 Ollama 시도, 실패 시 로컬 MLX
    if check_ollama_available():
        res = generate_ollama(TUNED_MODEL, prompt)
        if res is not None:
            return res
    
    return generate_mlx(prompt)

def clean_json(text):
    text = text.strip()
    # Markdown 코드 블록 제거
    if text.startswith("```json"):
        text = text[7:]
    elif text.startswith("```"):
        text = text[3:]
    if text.endswith("```"):
        text = text[:-3]
    text = text.strip()
    # 첫번째 {와 마지막 }를 찾아서 추출
    match = re.search(r"(\{.*\})", text, re.DOTALL)
    if match:
        return match.group(1)
    return text

def parse_to_json(text):
    try:
        cleaned = clean_json(text)
        return json.loads(cleaned)
    except Exception:
        return None

def run_evaluation():
    if not os.path.exists(GOLDEN_PATH):
        print(f"오류: 골든 데이터셋 파일이 존재하지 않습니다: {GOLDEN_PATH}")
        return

    print("골든 데이터셋을 로드하는 중...")
    with open(GOLDEN_PATH, "r", encoding="utf-8") as f:
        samples = [json.loads(line) for line in f if line.strip()]

    goal_samples = []
    mission_samples = []
    
    for s in samples:
        if "금융 목표 파싱" in s["prompt"]:
            goal_samples.append(s)
        else:
            mission_samples.append(s)

    print(f"로드 완료: 총 {len(samples)}건 (목표 파싱: {len(goal_samples)}건, 미션 생성: {len(mission_samples)}건)")

    results = {
        "base": {"goal_success": 0, "goal_json_ok": 0, "goal_item_ok": 0, "goal_price_mape": [],
                 "mission_success": 0, "mission_json_ok": 0, "mission_cat_ok": 0, "mission_numeric_ok": 0},
        "tuned": {"goal_success": 0, "goal_json_ok": 0, "goal_item_ok": 0, "goal_price_mape": [],
                  "mission_success": 0, "mission_json_ok": 0, "mission_cat_ok": 0, "mission_numeric_ok": 0}
    }

    # --- 1단계: Base Model 평가 패스 (VRAM 스왑 방지용 순차 연산) ---
    print("\n--- [Base Model] 평가 진행 중 ---")
    
    # Base 모델 - 목표 파싱 평가
    for idx, sample in enumerate(goal_samples):
        prompt = sample["prompt"]
        answer = json.loads(sample["completion"])
        base_res = query_model(BASE_MODEL, prompt)
        base_json = parse_to_json(base_res)
        if base_json:
            results["base"]["goal_json_ok"] += 1
            if base_json.get("itemName") == answer["itemName"]:
                results["base"]["goal_item_ok"] += 1
            ans_price = answer["price"]
            pred_price = base_json.get("price")
            if pred_price is None: pred_price = 0
            try: pred_price = int(pred_price)
            except Exception: pred_price = 0
            if ans_price > 0:
                mape = abs(ans_price - pred_price) / ans_price
                results["base"]["goal_price_mape"].append(mape)
            else:
                results["base"]["goal_price_mape"].append(0.0 if pred_price == 0 else 1.0)
        if (idx + 1) % 10 == 0:
            print(f"[Base Model] 목표 파싱 평가 진행률: {idx + 1}/{len(goal_samples)}")

    # Base 모델 - 미션 생성 평가
    for idx, sample in enumerate(mission_samples):
        prompt = sample["prompt"]
        answer = json.loads(sample["completion"])
        base_res = query_model(BASE_MODEL, prompt)
        base_json = parse_to_json(base_res)
        if base_json:
            results["base"]["mission_json_ok"] += 1
            base_cat = base_json.get("restraint", {}).get("categoryName")
            ans_cat = answer.get("restraint", {}).get("categoryName")
            if base_cat == ans_cat:
                results["base"]["mission_cat_ok"] += 1
            base_limit = base_json.get("restraint", {}).get("limitAmount", 0)
            base_transfer = base_json.get("saving", {}).get("transferAmount", 0)
            if isinstance(base_limit, (int, float)) and base_limit > 0 and isinstance(base_transfer, (int, float)) and base_transfer > 0:
                results["base"]["mission_numeric_ok"] += 1
        if (idx + 1) % 10 == 0:
            print(f"[Base Model] 미션 생성 평가 진행률: {idx + 1}/{len(mission_samples)}")


    # --- 2단계: Tuned Model 평가 패스 (VRAM 스왑 방지용 순차 연산) ---
    print("\n--- [Tuned Model] 평가 진행 중 ---")

    # Tuned 모델 - 목표 파싱 평가
    for idx, sample in enumerate(goal_samples):
        prompt = sample["prompt"]
        answer = json.loads(sample["completion"])
        tuned_res = query_model(TUNED_MODEL, prompt)
        tuned_json = parse_to_json(tuned_res)
        if tuned_json:
            results["tuned"]["goal_json_ok"] += 1
            if tuned_json.get("itemName") == answer["itemName"]:
                results["tuned"]["goal_item_ok"] += 1
            ans_price = answer["price"]
            pred_price = tuned_json.get("price")
            if pred_price is None: pred_price = 0
            try: pred_price = int(pred_price)
            except Exception: pred_price = 0
            if ans_price > 0:
                mape = abs(ans_price - pred_price) / ans_price
                results["tuned"]["goal_price_mape"].append(mape)
            else:
                results["tuned"]["goal_price_mape"].append(0.0 if pred_price == 0 else 1.0)
        if (idx + 1) % 10 == 0:
            print(f"[Tuned Model] 목표 파싱 평가 진행률: {idx + 1}/{len(goal_samples)}")

    # Tuned 모델 - 미션 생성 평가
    for idx, sample in enumerate(mission_samples):
        prompt = sample["prompt"]
        answer = json.loads(sample["completion"])
        tuned_res = query_model(TUNED_MODEL, prompt)
        tuned_json = parse_to_json(tuned_res)
        if tuned_json:
            results["tuned"]["mission_json_ok"] += 1
            tuned_cat = tuned_json.get("restraint", {}).get("categoryName")
            ans_cat = answer.get("restraint", {}).get("categoryName")
            if tuned_cat == ans_cat:
                results["tuned"]["mission_cat_ok"] += 1
            tuned_limit = tuned_json.get("restraint", {}).get("limitAmount", 0)
            tuned_transfer = tuned_json.get("saving", {}).get("transferAmount", 0)
            if isinstance(tuned_limit, (int, float)) and tuned_limit > 0 and isinstance(tuned_transfer, (int, float)) and tuned_transfer > 0:
                results["tuned"]["mission_numeric_ok"] += 1
        if (idx + 1) % 10 == 0:
            print(f"[Tuned Model] 미션 생성 평가 진행률: {idx + 1}/{len(mission_samples)}")

    # 결과 통계 요약 작성
    total_goals = len(goal_samples)
    total_missions = len(mission_samples)

    base_goal_json_rate = (results["base"]["goal_json_ok"] / total_goals) * 100
    base_goal_item_rate = (results["base"]["goal_item_ok"] / total_goals) * 100
    base_goal_mape = (sum(results["base"]["goal_price_mape"]) / len(results["base"]["goal_price_mape"])) * 100 if results["base"]["goal_price_mape"] else 100.0
    
    tuned_goal_json_rate = (results["tuned"]["goal_json_ok"] / total_goals) * 100
    tuned_goal_item_rate = (results["tuned"]["goal_item_ok"] / total_goals) * 100
    tuned_goal_mape = (sum(results["tuned"]["goal_price_mape"]) / len(results["tuned"]["goal_price_mape"])) * 100 if results["tuned"]["goal_price_mape"] else 100.0

    base_mission_json_rate = (results["base"]["mission_json_ok"] / total_missions) * 100
    base_mission_cat_rate = (results["base"]["mission_cat_ok"] / total_missions) * 100
    base_mission_num_rate = (results["base"]["mission_numeric_ok"] / total_missions) * 100
    
    tuned_mission_json_rate = (results["tuned"]["mission_json_ok"] / total_missions) * 100
    tuned_mission_cat_rate = (results["tuned"]["mission_cat_ok"] / total_missions) * 100
    tuned_mission_num_rate = (results["tuned"]["mission_numeric_ok"] / total_missions) * 100

    report_content = f"""## 2단계: Before/After 평가 결과 (골든 데이터셋 {len(samples)}건 기준)

### 1. 목표 파싱 (Goal Parsing) 평가 지표 (총 {total_goals}건)
| 지표 | 베이스 모델 ({BASE_MODEL}) | 파인튜닝 모델 ({TUNED_MODEL}) | 개선도 |
| :--- | :---: | :---: | :---: |
| **JSON 파싱 성공률** | {base_goal_json_rate:.1f}% | {tuned_goal_json_rate:.1f}% | {tuned_goal_json_rate - base_goal_json_rate:+.1f}%p |
| **물품명 일치율 (정확도)** | {base_goal_item_rate:.1f}% | {tuned_goal_item_rate:.1f}% | {tuned_goal_item_rate - base_goal_item_rate:+.1f}%p |
| **가격 평균 오차율 (MAPE)** | {base_goal_mape:.1f}% | {tuned_goal_mape:.1f}% | {base_goal_mape - tuned_goal_mape:+.1f}%p 개선 |

### 2. 미션 생성 (Mission Generation) 평가 지표 (총 {total_missions}건)
| 지표 | 베이스 모델 ({BASE_MODEL}) | 파인튜닝 모델 ({TUNED_MODEL}) | 개선도 |
| :--- | :---: | :---: | :---: |
| **JSON 파싱 성공률** | {base_mission_json_rate:.1f}% | {tuned_mission_json_rate:.1f}% | {tuned_mission_json_rate - base_mission_json_rate:+.1f}%p |
| **절제 카테고리 일치율** | {base_mission_cat_rate:.1f}% | {tuned_mission_cat_rate:.1f}% | {tuned_mission_cat_rate - base_mission_cat_rate:+.1f}%p |
| **미션 금액 수치 유효성** | {base_mission_num_rate:.1f}% | {tuned_mission_num_rate:.1f}% | {tuned_mission_num_rate - base_mission_num_rate:+.1f}%p |

* 가격 평균 오차율(MAPE)은 낮을수록 우수합니다.
* 미션 금액 수치 유효성은 `limitAmount` 및 `transferAmount`가 양의 정수로 생성되었는지를 검증합니다.
"""
    # 결과를 파일로 일단 임시 저장
    with open(os.path.join(SCRIPT_DIR, "eval_temp.md"), "w", encoding="utf-8") as f:
        f.write(report_content)
        
    print("\n=== Before/After 평가 완료 및 기록 완료 ===")
    print(report_content)

if __name__ == "__main__":
    run_evaluation()
